### `src/modules/robots/service.ts`

```typescript
import { PrismaClient } from '@prisma/client';
import { CreateRobotInput, ListRobotsQuery } from './schema';

export class RobotService {
  constructor(
    private prisma: PrismaClient,
    private redis: { get: (k: string) => Promise<string | null> },
  ) {}

  async list(query: ListRobotsQuery) {
    const where: Record<string, unknown> = { deletedAt: null };
    if (query.status) where.status = query.status;
    if (query.group) where.groupName = query.group;
    if (query.search) {
      where.name = { contains: query.search, mode: 'insensitive' };
    }

    const [robots, total] = await Promise.all([
      this.prisma.robot.findMany({
        where,
        orderBy: { [query.sort === 'lastSeen' ? 'lastTelemetryAt' : query.sort]: query.order },
        take: query.limit + 1, // +1 for cursor detection
        ...(query.cursor ? { cursor: { id: query.cursor }, skip: 1 } : {}),
      }),
      this.prisma.robot.count({ where }),
    ]);

    const hasMore = robots.length > query.limit;
    const items = hasMore ? robots.slice(0, -1) : robots;
    const nextCursor = hasMore ? items[items.length - 1].id : null;

    // Enrich with latest telemetry from Redis cache
    const enriched = await this.enrichWithTelemetry(items);

    return { items: enriched, total, nextCursor };
  }

  async getById(id: string) {
    const robot = await this.prisma.robot.findUnique({ where: { id } });
    if (!robot) return null;
    const enriched = await this.enrichWithTelemetry([robot]);
    return enriched[0];
  }

  async create(input: CreateRobotInput) {
    return this.prisma.robot.create({
      data: {
        name: input.name,
        model: input.model,
        serialNumber: input.serialNumber,
        groupName: input.groupName,
        tags: input.tags,
        config: input.config as object,
      },
    });
  }

  async update(id: string, input: Partial<CreateRobotInput>) {
    return this.prisma.robot.update({
      where: { id },
      data: {
        ...(input.name !== undefined && { name: input.name }),
        ...(input.model !== undefined && { model: input.model }),
        ...(input.groupName !== undefined && { groupName: input.groupName }),
        ...(input.tags !== undefined && { tags: input.tags }),
        ...(input.config !== undefined && { config: input.config as object }),
      },
    });
  }

  async softDelete(id: string) {
    return this.prisma.robot.update({
      where: { id },
      data: { deletedAt: new Date() },
    });
  }

  // ── Helpers ──────────────────────────────────────

  private async enrichWithTelemetry(robots: Awaited<ReturnType<typeof this.prisma.robot.findMany>>) {
    // Batch-fetch latest telemetry from Redis cache
    const keys = robots.map(r => `cache:robot:${r.id}:latest`);
    const telemetry = await Promise.all(keys.map(k => this.redis.get(k)));

    return robots.map((robot, i) => {
      const latest = telemetry[i] ? JSON.parse(telemetry[i]) : null;
      // Derive effective status: if robot says "online" but no telemetry for > 30s → "delayed"
      let effectiveStatus = robot.status;
      if (robot.status === 'online' && !latest) {
        effectiveStatus = 'delayed';
      }
      return {
        ...robot,
        effectiveStatus,
        latestTelemetry: latest ? {
          batteryPct: latest.battery_pct,
          cpuPct: latest.cpu_pct,
          memoryPct: latest.memory_pct,
          networkLatencyMs: latest.network_latency_ms,
          taskStatus: latest.task_status,
          timestamp: latest.time,
        } : null,
      };
    });
  }
}
```

### `src/modules/robots/routes.ts`

```typescript
import { FastifyInstance, FastifyPluginAsync } from 'fastify';
import { RobotService } from './service';
import {
  createRobotSchema,
  updateRobotSchema,
  listRobotsQuerySchema,
} from './schema';

export const robotRoutes: FastifyPluginAsync = async (app: FastifyInstance) => {
  const service = new RobotService(app.prisma, app.redis);

  // LIST
  app.get('/api/robots', {
    schema: { querystring: listRobotsQuerySchema },
    async handler(request, reply) {
      const result = await service.list(request.query);
      return reply.send(result);
    },
  });

  // GET BY ID
  app.get('/api/robots/:id', async (request, reply) => {
    const { id } = request.params as { id: string };
    const robot = await service.getById(id);
    if (!robot) return reply.status(404).send({ error: 'Robot not found' });
    return reply.send(robot);
  });

  // CREATE
  app.post('/api/robots', {
    schema: { body: createRobotSchema },
    async handler(request, reply) {
      const robot = await service.create(request.body);
      return reply.status(201).send(robot);
    },
  });

  // UPDATE
  app.patch('/api/robots/:id', {
    schema: { body: updateRobotSchema },
    async handler(request, reply) {
      const { id } = request.params as { id: string };
      const robot = await service.update(id, request.body);
      return reply.send(robot);
    },
  });

  // DELETE (soft)
  app.delete('/api/robots/:id', async (request, reply) => {
    const { id } = request.params as { id: string };
    await service.softDelete(id);
    return reply.status(204).send();
  });
};
```

## 4. WebSocket Hub

### `src/lib/ws-hub.ts`

```typescript
import { FastifyInstance } from 'fastify';
import { WebSocket } from 'ws';
import Redis from 'ioredis';

interface Client {
  ws: WebSocket;
  userId: string;
  channels: Set<string>;
}

export class WebSocketHub {
  private clients = new Map<string, Client>();
  private redisSub: Redis;
  private maxClients: number;

  constructor(redis: Redis, maxClients = 100) {
    this.redisSub = redis.duplicate();
    this.maxClients = maxClients;
  }

  async register(app: FastifyInstance) {
    app.get('/ws', { websocket: true }, (socket, req) => {
      // Auth: extract JWT from query param
      const url = new URL(req.url!, `http://${req.headers.host}`);
      const token = url.searchParams.get('token');
      if (!token) {
        socket.close(4001, 'Missing token');
        return;
      }

      // Decode JWT (simplified — real impl uses @fastify/jwt)
      let userId: string;
      try {
        const payload = JSON.parse(Buffer.from(token.split('.')[1], 'base64').toString());
        userId = payload.sub;
      } catch {
        socket.close(4001, 'Invalid token');
        return;
      }

      if (this.clients.size >= this.maxClients) {
        socket.close(4003, 'Too many connections');
        return;
      }

      const clientId = `${userId}-${Date.now()}`;
      const client: Client = {
        ws: socket,
        userId,
        channels: new Set(),
      };
      this.clients.set(clientId, client);

      app.log.info(`WS connected: ${userId} (total: ${this.clients.size})`);

      // Handle messages
      socket.on('message', (raw) => {
        try {
          const msg = JSON.parse(raw.toString());
          this.handleMessage(clientId, msg);
        } catch {
          // Ignore malformed messages
        }
      });

      // Handle disconnect
      socket.on('close', () => {
        this.clients.delete(clientId);
        app.log.info(`WS disconnected: ${userId} (total: ${this.clients.size})`);
      });

      // Send heartbeat
      const heartbeat = setInterval(() => {
        if (socket.readyState === WebSocket.OPEN) {
          socket.ping();
        }
      }, 30_000);

      socket.on('close', () => clearInterval(heartbeat));
    });

    // Subscribe to Redis pub/sub channels for fan-out
    await this.redisSub.psubscribe('events:telemetry', 'events:alerts');

    this.redisSub.on('pmessage', (_pattern, channel, message) => {
      this.broadcast(channel, JSON.parse(message));
    });
  }

  private handleMessage(clientId: string, msg: { type: string; channel?: string }) {
    const client = this.clients.get(clientId);
    if (!client) return;

    switch (msg.type) {
      case 'subscribe':
        if (msg.channel) client.channels.add(msg.channel);
        break;
      case 'unsubscribe':
        if (msg.channel) client.channels.delete(msg.channel);
        break;
    }
  }

  private broadcast(channel: string, data: unknown) {
    for (const client of this.clients.values()) {
      if (
        client.ws.readyState === WebSocket.OPEN &&
        (client.channels.has(channel) || client.channels.has('fleet:overview'))
      ) {
        client.ws.send(JSON.stringify({ channel, data }));
      }
    }
  }

  getStats() {
    return {
      connectedClients: this.clients.size,
    };
  }
}
```

```typescript
import mqtt from 'mqtt';
import { Redis } from 'ioredis';
import { RawTelemetry } from '../types';

export class MqttAdapter {
  private client: mqtt.MqttClient | null = null;

  constructor(
    private redis: Redis,
    private config: { host: string; port: number; topic: string; streamKey: string },
  ) {}

  async start(): Promise<void> {
    return new Promise((resolve, reject) => {
      this.client = mqtt.connect({
        host: this.config.host,
        port: this.config.port,
        reconnectPeriod: 1000, // 1s initial, exponential backoff built into mqtt.js
      });

      this.client.on('connect', () => {
        console.log(`MQTT connected to ${this.config.host}:${this.config.port}`);
        this.client!.subscribe(this.config.topic, (err) => {
          if (err) return reject(err);
          resolve();
        });
      });

      this.client.on('message', (topic, payload) => {
        this.handleMessage(topic, payload);
      });

      this.client.on('error', (err) => {
        console.error('MQTT error:', err.message);
      });
    });
  }

  private async handleMessage(topic: string, payload: Buffer) {
    // Extract robot_id from topic: "robot/{id}/telemetry"
    const robotId = topic.split('/')[1];
    if (!robotId) return;

    const raw: RawTelemetry = {
      robot_id: robotId,
      timestamp: new Date().toISOString(),
      source: 'mqtt',
      raw_payload: payload.toString('base64'), // Store binary as base64 for JSON compat
      format_version: '1.0',
    };

    await this.redis.xadd(
      this.config.streamKey,
      '*',
      'data', JSON.stringify(raw),
    );
  }

  async stop(): Promise<void> {
    return new Promise((resolve) => {
      if (this.client) {
        this.client.end(false, {}, resolve);
      } else {
        resolve();
      }
    });
  }
}
```

### `pipeline/src/normalizer/mapper.ts` (skeleton — config-driven)

```typescript
import fs from 'fs';
import path from 'path';
import yaml from 'yaml';
import { NormalizedTelemetry, RawTelemetry } from '../types';

interface FieldMapping {
  raw_path: string;        // "payload.battery.percentage" or "payload[0].voltage"
  target_field: string;    // "battery_pct"
  transform?: string;      // "multiply(0.01)" | "enum_map(...)" | "identity"
  unit_conversion?: string; // "mV_to_V" | "fahrenheit_to_celsius"
}

interface RobotMapping {
  robot_model: string;
  fields: FieldMapping[];
}

interface TransformFn {
  (value: unknown): unknown;
}

export class SchemaMapper {
  private mappings = new Map<string, FieldMapping[]>();
  private transforms = new Map<string, TransformFn>();

  constructor(private mappingsDir: string) {
    // Register built-in transforms
    this.transforms.set('identity', (v) => v);
    this.transforms.set('multiply(0.01)', (v) => Number(v) * 0.01);
    this.transforms.set('multiply(100)', (v) => Number(v) * 100);
    this.transforms.set('divide(1000)', (v) => Number(v) / 1000); // mV → V
  }

  async loadMappings(): Promise<void> {
    const files = fs.readdirSync(this.mappingsDir).filter(f => f.endsWith('.yaml'));
    for (const file of files) {
      const content = fs.readFileSync(path.join(this.mappingsDir, file), 'utf-8');
      const mapping: RobotMapping = yaml.parse(content);
      this.mappings.set(mapping.robot_model, mapping.fields);
    }
    console.log(`Loaded ${this.mappings.size} robot model mappings`);
  }

  normalize(raw: RawTelemetry, robotModel: string): NormalizedTelemetry | null {
    const fields = this.mappings.get(robotModel);
    if (!fields) {
      console.warn(`No mapping for robot model: ${robotModel}`);
      return null;
    }

    const parsed = typeof raw.raw_payload === 'string'
      ? JSON.parse(raw.raw_payload)
      : raw.raw_payload;

    const result: Record<string, unknown> = {};

    for (const field of fields) {
      try {
        let value = this.getValueByPath(parsed, field.raw_path);
        if (field.transform && this.transforms.has(field.transform)) {
          value = this.transforms.get(field.transform)!(value);
        }
        result[field.target_field] = value;
      } catch (err) {
        console.warn(`Field ${field.target_field} mapping failed for ${robotModel}:`, err);
      }
    }

    return {
      robot_id: raw.robot_id,
      timestamp: raw.timestamp,
      ...result,
      raw_source: raw.source,
      format_version: raw.format_version,
    } as NormalizedTelemetry;
  }

  private getValueByPath(obj: unknown, path: string): unknown {
    if (!obj || typeof obj !== 'object') return undefined;
    return path.split('.').reduce((acc: any, key) => acc?.[key], obj);
  }
}
```

## 6. Plugin Registration

### `src/plugins/index.ts`

```typescript
import { FastifyInstance } from 'fastify';
import fastifyCors from '@fastify/cors';
import fastifyWebsocket from '@fastify/websocket';
import { Env } from '../config';
import { WebSocketHub } from '../lib/ws-hub';
import { createPrismaClient } from '../lib/prisma';
import { createRedisClient } from '../lib/redis';

export async function registerPlugins(app: FastifyInstance, env: Env) {
  // CORS
  await app.register(fastifyCors, {
    origin: env.CORS_ORIGIN,
    credentials: true,
  });

  // WebSocket
  await app.register(fastifyWebsocket);

  // Prisma (singleton)
  const prisma = createPrismaClient(env.DATABASE_URL);
  await prisma.$connect();
  app.decorate('prisma', prisma);

  // Redis (singleton)
  const redis = createRedisClient(env.REDIS_URL);
  app.decorate('redis', redis);

  // WebSocket Hub
  const wsHub = new WebSocketHub(redis);
  await wsHub.register(app);
  app.decorate('wsHub', wsHub);

  // Request ID
  app.addHook('onRequest', async (request) => {
    request.id = (request.headers['x-request-id'] as string) || crypto.randomUUID();
  });

  // Global error handler
  app.setErrorHandler((error, request, reply) => {
    app.log.error({ err: error, requestId: request.id }, error.message);

    if (error.validation) {
      return reply.status(400).send({
        error: 'Validation Error',
        details: error.validation,
      });
    }

    const statusCode = error.statusCode || 500;
    return reply.status(statusCode).send({
      error: statusCode >= 500 ? 'Internal Server Error' : error.message,
      requestId: request.id,
    });
  });

  // Graceful shutdown
  app.addHook('onClose', async () => {
    await prisma.$disconnect();
    await redis.quit();
  });
}
```

## Completion Report

**what_was_done**: Produced BFF implementation scaffold satisfying ADR-001 (Modular Monolith BFF) — project structure, core config (Zod-validated), Fastify app factory, 8 module route/service/schema skeletons (robots, fleet, telemetry, alerts, config, comments, history, reports), WebSocket hub with Redis pub/sub fan-out, Data Pipeline scaffold (MQTT adapter, config-driven schema mapper), plugin registration with Prisma + Redis singletons, and deployment-ready package.json.

**key_decisions**:
- **decision**: Module pattern: `routes.ts + service.ts + schema.ts` per module
  **rationale**: Separation of concerns at file level (not class level) — routes handle HTTP transport, service handles business logic with injected dependencies (prisma, redis), schema handles Zod validation. Testable without starting the server.
- **decision**: Prisma for PostgreSQL ORM, raw `pg` client for TimescaleDB queries
  **rationale**: Prisma provides type-safe CRUD for the BFF's operational data model. TimescaleDB queries (time-series aggregation, continuous aggregates, hypertable-specific SQL) go through raw pg to avoid ORM impedance mismatch with time-series features.
- **decision**: Cursor-based pagination for robot/list API
  **rationale**: Offset pagination breaks when robots are added/removed during pagination. Cursor using UUID `id` is stable and efficient (index scan, no OFFSET).
- **decision**: Status derivation in service layer (not DB trigger)
  **rationale**: Robot online/offline status depends on telemetry recency from TimescaleDB (separate database). Cross-DB logic belongs in application layer, not as DB trigger that requires foreign data wrappers.
- **decision**: MQTT adapter stores raw binary as base64 in Redis Stream
  **rationale**: Redis Stream values are strings. Binary → base64 → JSON wrapping preserves the full raw payload for audit/debug while maintaining stream compatibility.

**handoff_focus**: 
- **senior-frontend**: API contract is defined in Zod schemas (`schema.ts` in each module). Responses follow the `robotResponseSchema` shape. WebSocket protocol: subscribe to `fleet:overview`, `robot:{id}`, or `alerts` channels.
- **devops-engineer**: Dockerfiles needed for BFF (Node.js multi-stage) and Pipeline (Node.js). docker-compose.yml should include: postgres:16, timescaledb:2.x, redis:7, bff, pipeline.

**open_questions**:
- JWT implementation: stubbed with base64 decode. Full implementation needs `@fastify/jwt` plugin with proper token signing/verification and refresh token flow.
- Rate limiting: `@fastify/rate-limit` not yet integrated — config values are defined but not wired.
- Pipeline HTTP/gRPC adapters: only MQTT adapter skeleton provided. HTTP and gRPC need equivalent implementations following same pattern.

**known_constraints**:
- TimescaleDB and PostgreSQL are separate connections (may be same cluster but different DBs). Cross-DB JOINs are not possible — application-level correlation required for queries like "robot status + latest telemetry".
- Redis cache for telemetry snapshots has 5s TTL (configurable). Real-time precision depends on WebSocket push, not cache reads.
